\subsection*{Problema 02}

\noindent\textbf{Enunciado:} Evaluar la integral definida 
mediante sumas de Riemann usando la partición 
regular:
\begin{align*}
\int_{0}^{3} x \, dx
\end{align*}

\noindent\textbf{Solución:} 
Aplicamos la definición de la integral definida como 
el límite de una suma de Riemann por la derecha:
\begin{align*}
\int_{a}^{b} f(x) \, dx = \lim_{n \to \infty} \sum_{i=1}^{n} f(x_i) \Delta x
\end{align*}

Identificamos los límites de integración $a = 0$ y 
$b = 3$, y la función $f(x) = x$. Definimos el ancho 
de cada subintervalo $\Delta x$ y los puntos de 
evaluación $x_i$ según las indicaciones:
\begin{align*}
\Delta x &= \dfrac{3}{n}, \quad 
x_i = 0 + i \Delta x = \dfrac{3i}{n}
\end{align*}

Evaluamos la función en los puntos $x_i$:
\begin{align*}
f(x_i) = f\left(\dfrac{3i}{n}\right) = \dfrac{3i}{n}
\end{align*}

Sustituimos estos términos en la suma de Riemann:
\begin{align*}
\sum_{i=1}^{n} f(x_i) \Delta x &= \sum_{i=1}^{n} 
\left(\dfrac{3i}{n}\right) \left(\dfrac{3}{n}\right) \\
&= \sum_{i=1}^{n} \dfrac{9i}{n^2}
\end{align*}

Extraemos las constantes fuera del operador sumatorio 
y aplicamos la fórmula para la suma de los primeros 
$n$ enteros ($\sum i = \dfrac{n(n+1)}{2}$):
\begin{align*}
\sum_{i=1}^{n} \dfrac{9i}{n^2} &= \dfrac{9}{n^2} 
\sum_{i=1}^{n} i \\
&= \dfrac{9}{n^2} \left[ \dfrac{n(n+1)}{2} \right]
\end{align*}

Simplificamos la expresión algebraica resultante:
\begin{align*}
\dfrac{9n(n+1)}{2n^2} &= \dfrac{9}{2} \left( \dfrac{n+1}{n} 
\right) \\
&= \dfrac{9}{2} \left( 1 + \dfrac{1}{n} \right)
\end{align*}

Calculamos el límite cuando $n$ tiende al infinito 
para obtener el valor exacto de la integral:
\begin{align*}
\int_{0}^{3} x \, dx &= \lim_{n \to \infty} \dfrac{9}{2} 
\left( 1 + \dfrac{1}{n} \right) \\
&= \dfrac{9}{2} (1 + 0) \\
&= \dfrac{9}{2} = 4.5
\end{align*}
